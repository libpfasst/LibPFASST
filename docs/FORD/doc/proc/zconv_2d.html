  subroutine zconv_2d(this, ghat,op,chat)
    class(pf_fft_abs_t), intent(inout) :: this
    complex(pfdp), intent(in) :: ghat(:,:),op(:,:)
    complex(pfdp), intent(inout) ::chat(:,:)


    this%wk_2d=ghat
    call this%fftb()
    this%wk_2d = this%wk_2d * op
    call this%fftf()
    chat=this%wk_2d


  end subroutine zconv_2d
