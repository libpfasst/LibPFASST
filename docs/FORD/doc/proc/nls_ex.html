public function nls_ex(t, x, Lx) result(u)

Routine to return the exact solution for the nonlinear Schoedinger Eq u_t=i2}u|^2u +i u_xx See Tuncay Aktosun et al 2007 Inverse Problems 23 217 Note the domain size should be 2*pi

  function nls_ex(t, x,Lx) result(u)
    real(pfdp), intent(in)  :: t
    real(pfdp), intent(in)  :: x
    real(pfdp), intent(in)  :: Lx        
    complex(pfdp)  :: u

    real(pfdp) :: a,b,taa,c
    complex(pfdp) :: ae,dn

    a=1.0_pfdp/sqrt(2.0_pfdp)
    b=1.0_pfdp
    taa=1.0_pfdp
    ae=a*exp(zi*t)
    dn=cosh(t) + zi*sinh(t)
    c = 4.0_pfdp
       
    u=exp(zi*c*(x-c*t))*ae*(dn/(cosh(t)-sqrt(2.0_pfdp)/2.0_pfdp*cos(x-2.0_pfdp*c*t))-1.0_pfdp)

  end function nls_ex
